export function deviceGroups(): { group: string; items: DevicePreset[] }[] {
  const order: string[] = [];
  const map = new Map<string, DevicePreset[]>();
  for (const d of DEVICE_PRESETS) {
    const g = d.group ?? "鍏朵粬";
    if (!map.has(g)) {
      map.set(g, []);
      order.push(g);
    }
    map.get(g)!.push(d);
  }
  return order.map((g) => ({ group: g, items: map.get(g)! }));
}
